const ROUTES = {
  MASTER_DATA: "master-data",
  PROCESS_IMAGE_AI: "process-image-ai",
  SAVE_PROJECT: "save-project",
  SAVED_PROJECTS: "saved-projects",
  PROJECT_DETAILS: "project-details",
  VERIFY_PASSWORD: "verify-password",
  DELETE_PROJECT: "delete-project",
  HEALTH: "health",
  LINE_INTAKE: "line-intake",
  LINE_SEARCH: "line-search"
};

function doGet(e) {
  return handleApiRequest_(e, "GET");
}

function doPost(e) {
  return handleApiRequest_(e, "POST");
}

function handleApiRequest_(e, method) {
  try {
    const params = (e && e.parameter) || {};
    const body = parseRequestBody_(e);
    const action = String(params.action || body.action || "").trim();

    switch (action) {
      case ROUTES.HEALTH:
        return jsonOutput_({
          ok: true,
          route: action,
          method: method,
          date: new Date().toISOString()
        });

      case ROUTES.MASTER_DATA:
        assertMethod_(method, "GET");
        return jsonOutput_(getMasterData());

      case ROUTES.SAVED_PROJECTS:
        assertMethod_(method, "GET");
        return jsonOutput_(getSavedProjects());

      case ROUTES.PROCESS_IMAGE_AI:
        assertMethod_(method, "POST");
        return jsonOutput_(processImageAI(body.base64Data, body.contentType));

      case ROUTES.SAVE_PROJECT:
        assertMethod_(method, "POST");
        return jsonOutput_(saveProject(body));

      case ROUTES.PROJECT_DETAILS:
        assertMethod_(method, "POST");
        return jsonOutput_(getProjectDetails(body.projectId));

      case ROUTES.VERIFY_PASSWORD:
        assertMethod_(method, "POST");
        return jsonOutput_(verifyPassword(body.password));

      case ROUTES.DELETE_PROJECT:
        assertMethod_(method, "POST");
        return jsonOutput_(deleteProject(body.projectId, body.password));

      case ROUTES.LINE_INTAKE:
        assertMethod_(method, "POST");
        return jsonOutput_(createLineIntake(body));

      case ROUTES.LINE_SEARCH:
        assertMethod_(method, "POST");
        return jsonOutput_(searchProjectsForLine(body.query));

      default:
        return jsonOutput_({
          error: true,
          message: "Unknown action",
          allowedRoutes: Object.keys(ROUTES).map(function (key) { return ROUTES[key]; })
        });
    }
  } catch (error) {
    return jsonOutput_({
      error: true,
      message: error.message || String(error)
    });
  }
}

function assertMethod_(actualMethod, expectedMethod) {
  if (actualMethod !== expectedMethod) {
    throw new Error("Method not allowed. Expected " + expectedMethod + " but received " + actualMethod);
  }
}

function parseRequestBody_(e) {
  try {
    if (!e || !e.postData || !e.postData.contents) return {};
    return JSON.parse(e.postData.contents);
  } catch (error) {
    throw new Error("Invalid JSON request body");
  }
}

function jsonOutput_(data) {
  return ContentService
    .createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

function processImageAI(base64Data, contentType) {
  const modelName = "gemini-2.5-flash";
  const url = "https://generativelanguage.googleapis.com/v1beta/models/" + modelName + ":generateContent?key=" + GEMINI_API_KEY;

  const prompt = "วิเคราะห์ภาพบัญชีวัสดุ กฟภ. (Bill of Materials):\n" +
    "1. สกัดรหัสพัสดุ (MAT. NO.): ให้สกัดข้อความทั้งหมดที่ปรากฏในช่องรหัสพัสดุมาเป็นค่า 'id' เพียงก้อนเดียวใน 1 object (เช่น '1000010004 or 1000010012', '101010/102020' หรือ '1050010200-4') ห้ามแยกเป็นหลายรายการเด็ดขาด\n" +
    "2. สกัดจำนวน (qty): ให้ดูตัวเลขจากคอลัมน์ REQ'D คอลัมน์แรก (ช่อง I) เพียงช่องเดียวเท่านั้น\n" +
    "   - ห้ามนำตัวเลขจากคอลัมน์อื่น (เช่น II, III, IV) มาบวกกันโดยเด็ดขาด\n" +
    "   - หากคอลัมน์แรก (ช่อง I) เป็นช่องว่างหรือไม่มีตัวเลข ให้กำหนดค่า qty เป็น 0\n" +
    "3. ตอบเป็น JSON Array เท่านั้น: [{\"id\":\"...\",\"qty\":...}]";

  const payload = {
    contents: [{
      parts: [
        { text: prompt },
        { inline_data: { mime_type: contentType, data: base64Data } }
      ]
    }],
    generationConfig: {
      response_mime_type: "application/json"
    }
  };

  try {
    const response = UrlFetchApp.fetch(url, {
      method: "POST",
      contentType: "application/json",
      payload: JSON.stringify(payload),
      muteHttpExceptions: true
    });
    const result = JSON.parse(response.getContentText());
    if (result.error) throw new Error("Google API: " + result.error.message);
    const aiText = result.candidates[0].content.parts[0].text;
    return JSON.parse(aiText);
  } catch (e) {
    return { error: true, msg: "เกิดข้อผิดพลาด: " + e.toString() };
  }
}

function getMasterData() {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet = ss.getSheetByName("2026-1");
    const data = sheet.getDataRange().getValues();
    const laborSheet = ss.getSheetByName("ค่าแรง");
    const laborData = laborSheet ? laborSheet.getDataRange().getValues() : [];

    const laborMap = {};
    if (laborData.length > 0) {
      laborData.slice(1).forEach(function (r) {
        const matId = String(r[0]);
        if (!laborMap[matId]) laborMap[matId] = [];
        laborMap[matId].push({
          desc: String(r[2] || "ทั่วไป"),
          price: parseFloat(r[4]) || 0
        });
      });
    }

    return data.slice(1).map(function (r) {
      const id = String(r[1] || "");
      const options = laborMap[id] || [{
        desc: "ค่าแรงมาตรฐาน",
        price: parseFloat(r[5]) || 0
      }];

      return {
        id: id,
        name: String(r[2] || ""),
        unit: String(r[3] || ""),
        matPrice: parseFloat(r[4]) || 0,
        labPrice: options[0].price,
        laborOptions: options
      };
    }).filter(function (item) {
      return item.id !== "" || item.name !== "";
    });
  } catch (e) {
    return [];
  }
}

function saveProject(payload) {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const dbSheet = ss.getSheetByName("Project_Database") || ss.insertSheet("Project_Database");
    const detSheet = ss.getSheetByName("Project_Details") || ss.insertSheet("Project_Details");
    const projectId = payload.projectId || "PJ-" + new Date().getTime();

    const parentFolder = DriveApp.getFolderById(FOLDER_ID);
    const folders = parentFolder.getFoldersByName(projectId);
    const projectFolder = folders.hasNext() ? folders.next() : parentFolder.createFolder(projectId);

    if (payload.fileList && payload.fileList.length > 0) {
      payload.fileList.forEach(function (fileData, index) {
        const timestamp = new Date().getTime();
        const blob = Utilities.newBlob(
          Utilities.base64Decode(fileData.base64),
          fileData.type,
          "Scan_" + projectId + "_" + timestamp + "_" + index
        );
        const file = projectFolder.createFile(blob);
        file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
      });
    }

    const allFiles = projectFolder.getFiles();
    const imageUrls = [];
    while (allFiles.hasNext()) {
      imageUrls.push(allFiles.next().getUrl());
    }
    const finalImageUrlStr = imageUrls.join("|");

    const summaryRow = [
      projectId,
      new Date(),
      payload.pjName,
      payload.grandTotal,
      finalImageUrlStr
    ];

    const dbData = dbSheet.getDataRange().getValues();
    let rowToUpdate = -1;
    for (let i = 1; i < dbData.length; i++) {
      if (dbData[i][0] === projectId) {
        rowToUpdate = i + 1;
        break;
      }
    }

    if (rowToUpdate !== -1) {
      dbSheet.getRange(rowToUpdate, 1, 1, summaryRow.length).setValues([summaryRow]);
      const detData = detSheet.getDataRange().getValues();
      for (let j = detData.length - 1; j >= 1; j--) {
        if (detData[j][0] === projectId) detSheet.deleteRow(j + 1);
      }
    } else {
      dbSheet.appendRow(summaryRow);
    }

    payload.budgets.forEach(function (budget) {
      budget.items.forEach(function (item) {
        detSheet.appendRow([
          projectId,
          budget.type,
          item.id,
          item.name,
          item.qty,
          item.total,
          item.laborDesc || "ค่าแรงมาตรฐาน",
          item.labPrice
        ]);
      });
    });

    return { status: "success", msg: "บันทึกเรียบร้อย", projectId: projectId };
  } catch (e) {
    return { status: "error", msg: e.toString() };
  }
}

function getSavedProjects() {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet = ss.getSheetByName("Project_Database");
    if (!sheet || sheet.getLastRow() < 2) return [];

    const data = sheet.getDataRange().getDisplayValues();
    return data.slice(1).reverse().map(function (r) {
      return [
        String(r[0]),
        String(r[1]),
        String(r[2]),
        r[3],
        String(r[4] || "")
      ];
    });
  } catch (e) {
    return [];
  }
}

function verifyPassword(password) {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const settings = ss.getSheetByName("Settings");
    return password === settings.getRange("B1").getValue().toString();
  } catch (e) {
    return false;
  }
}

function deleteProject(projectId, password) {
  if (!verifyPassword(password)) {
    return { status: "error", msg: "รหัสผ่านไม่ถูกต้อง" };
  }

  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  ["Project_Database", "Project_Details"].forEach(function (name) {
    const sheet = ss.getSheetByName(name);
    const data = sheet.getDataRange().getValues();
    for (let i = data.length - 1; i >= 1; i--) {
      if (data[i][0] === projectId) sheet.deleteRow(i + 1);
    }
  });

  return { status: "success", msg: "ลบสำเร็จ" };
}

function getProjectDetails(projectId) {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet = ss.getSheetByName("Project_Details");
    if (!sheet) return [];

    const data = sheet.getDataRange().getValues();
    return data.filter(function (r) {
      return r[0] === projectId;
    }).map(function (r) {
      return {
        type: r[1],
        id: String(r[2]),
        name: r[3],
        qty: r[4],
        total: r[5],
        laborDesc: r[6] || "ค่าแรงมาตรฐาน",
        labPrice: parseFloat(r[7]) || 0
      };
    });
  } catch (e) {
    return [];
  }
}

function createLineIntake(payload) {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet = ss.getSheetByName("Line_Intake_Queue") || ss.insertSheet("Line_Intake_Queue");

    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        "Created At",
        "Line User ID",
        "Display Name",
        "Source Type",
        "Message Text",
        "File URL",
        "Status",
        "Meta JSON"
      ]);
    }

    sheet.appendRow([
      new Date(),
      String(payload.userId || ""),
      String(payload.displayName || ""),
      String(payload.sourceType || "line"),
      String(payload.messageText || ""),
      String(payload.fileUrl || ""),
      "NEW",
      JSON.stringify(payload.meta || {})
    ]);

    return { status: "success", msg: "line intake queued" };
  } catch (e) {
    return { status: "error", msg: e.toString() };
  }
}

function searchProjectsForLine(query) {
  try {
    const term = String(query || "").trim().toLowerCase();
    if (!term) return [];

    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet = ss.getSheetByName("Project_Database");
    if (!sheet || sheet.getLastRow() < 2) return [];

    const data = sheet.getDataRange().getDisplayValues().slice(1);
    return data.filter(function (row) {
      const projectId = String(row[0] || "").toLowerCase();
      const projectName = String(row[2] || "").toLowerCase();
      return projectId.indexOf(term) !== -1 || projectName.indexOf(term) !== -1;
    }).slice(0, 10).map(function (row) {
      return {
        projectId: String(row[0] || ""),
        date: String(row[1] || ""),
        projectName: String(row[2] || ""),
        grandTotal: String(row[3] || ""),
        imageUrls: String(row[4] || "")
      };
    });
  } catch (e) {
    return [];
  }
}
